{%- comment -%} In-article AdSense units. Posts and content pages share this include and the in_article slot. mode defaults to page.ads, then "full". The page layout passes mode="full" so ads: light still gets the post pattern. ad.html renders nothing when page.ads is false. Units go before a later h2 on long pages. They stay out of a block that is still open: Key to Keys, a User's Guide chapter list (guide-series / chlist), a topic board, a card row, or the Explore catalog. {%- endcomment -%} {%- assign mode = include.mode | default: page.ads | default: "full" -%} {%- assign words = include.html | number_of_words -%} {%- assign parts = include.html | split: "= 1500 and n >= 6 -%} {%- assign a = n | divided_by: 3 -%}{%- assign b = n | times: 2 | divided_by: 3 -%} {%- elsif words >= 600 and n >= 3 -%} {%- assign a = n | plus: 1 | divided_by: 2 -%} {%- endif -%} {%- if a < 2 and a != -1 -%}{%- assign a = 2 -%}{%- endif -%} {%- endif -%} {%- assign placed_a = false -%} {%- assign so_far = "" -%} {%- assign block_markers = "key-to-keys|guide-series|chlist|topic-board|cols|explore" | split: "|" -%} {%- for part in parts -%} {%- if forloop.first -%}{{ part }}{%- assign so_far = part -%}{%- continue -%}{%- endif -%} {%- assign i = forloop.index0 -%} {%- assign prev_index = i | minus: 1 -%} {%- assign prev = parts[prev_index] -%} {%- assign risky = false -%} {%- if prev contains 'youtube' or prev contains '' | size | minus: 1 -%} {%- if open_divs >= close_divs -%}{%- assign inside_block = true -%}{%- endif -%} {%- endif -%} {%- endif -%} {%- endfor -%} {%- if inside_block == false and placed_a == false and a != -1 and i >= a and risky == false -%} {% include ad.html slot="in_article" %}{%- assign placed_a = true -%}{%- assign min_b = i | plus: 2 -%} {%- elsif inside_block == false and placed_a and b != -1 and i >= b and i >= min_b and risky == false -%} {% include ad.html slot="in_article" %}{%- assign b = -1 -%} {%- endif -%}
